void readBook(bookStore book[], int &noBook){
	// reads the text file and store the information in to the bookStore using struct array
	ifstream readFile("book.txt")
	if(readFile.fail())
		cout << "File has failed to open " << endl;
	else{
		while(!readFile.eof()){
			getline(readFile, book[noBook].title);
			getline(readFile, book[noBook].author);
			readFile >> book[noBook].year >> book[noBook].ranking;
