```cpp
bool reverseQueue(Queue *q){
	if (!q) return 0;
	Stack* s{new Stack()};
	int temp{};
	while(!(q->empty())){
		q->dequeue(temp);	
		s->push(temp);
	}
	while(!(s->empty())){
		s->pop(temp);
		q->enqueue(temp);
	}
	return 1;
}
```

int BinarySearchRec(int key, int list [], int start, int end){
	if end >= start{
		int middle = (start+ end) / 2
		if (list[middle] == key) return key;
		if (list[middle] > key) return BinarySearchRec(key, list, start, middle-1);
		return BinarySearchRec(key, list, middle+1, end);
	}
	return 1;
}
