import java.io.*;
public class Main{
	public static void main(String[] args) throws IOException{
		writeFile()
		try{
			readFile();
		}
		catch (IOException e){
			e.getMessage();
		}
	}
	public void writeFile(){
		FileOutputStream file = new FileOutputStream("temp.dat", true);
		DataOutputStream ofstream = new DataOutputStream(file);
		for(int index = 1; index <= 10; index++)
			DataOutputStream.writeInt(index);
	}
	public void readFile() throws IOException{
		FileInputStream file = new FileInputStream("temp.dat");
		if (!file.exists() || !file.canRead())
			throw new IOException("File cannot be read!");
		DataInputStream ifstream = new DataInputStream(file);
		boolean endOfFile = false;
		int number;
		
		while(!endOfFile){
			try{
				number = ifstream.readInt();
				System.out.print(number); 
			}
			catch(EOFException e){
				endOfFile = true;
			}
		}
	}
}
